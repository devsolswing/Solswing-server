    <h2>Game-server based, not browser-trusted</h2>
    <p class="lead">The original version decided everything in the browser — anyone could edit the rug time or fake a win. SolSwing flips that: the server is the referee, the Solana program is the vault.</p>
    <div class="flow">
      <div class="flow-step"><i>STEP 1 · ESCROW</i><h3>Both players stake</h3><p><span class="mono">initialize_match</span> (P1) + <span class="mono">join_match</span> (P2) lock equal wagers into a PDA vault. Two-phase, so nobody has to co-sign someone else's transaction.</p></div>

      <div class="flow-step"><i>STEP 3 · REFLEX RACE</i><h3>Fastest sell wins</h3><p>Server timestamps sell messages on arrival — client timestamps are ignored. Late sells (&gt;5s), double sells, disconnects and hidden-tab forfeits are all handled server-side.</p></div>
      <div class="flow-step"><i>STEP 4 · SETTLE ON-CHAIN</i><h3>Adjudicator pays out</h3><p>Only the adjudicator (game server key) can call <span class="mono">settle_match</span>. The program checks the winner, then pays 94% to the winner and 6% to the treasury with checked math.</p></div>

var WalletReadyState={ Installed:'Installed', NotDetected:'NotDetected', Loadable:'Loadable', Unsupported:'Unsupported' };
function phantomProvider(){
  if(window.phantom && window.phantom.solana && window.phantom.solana.isPhantom) return window.phantom.solana;
  if(window.solana && window.solana.isPhantom) return window.solana;
  return null;
}
function phantomInstalled(){ return !!phantomProvider(); }
// Phantom's extension (and its in-app browser) inject the provider
// ASYNCHRONOUSLY — it can land after DOMContentLoaded, after window load,
// even a beat after the player taps Connect. Any one-shot detection check
// is a race that loses on slow devices, so detection POLLS: resolves true
// the moment the provider appears, false only after the full timeout.
function waitForPhantom(ms){
  return new Promise(function(res){
    if(phantomInstalled()) return res(true);
    var t0=Date.now();
    var iv=setInterval(function(){
      if(phantomInstalled()){ clearInterval(iv); res(true); }
      else if(Date.now()-t0>=ms){ clearInterval(iv); res(false); }
    },120);
  });
}
function isMobileBrowser(){
  var ua=navigator.userAgent||'';
  return /Android|iPhone|iPad|iPod/i.test(ua) || (navigator.maxTouchPoints>1 && /Macintosh/.test(ua));
}
function isAndroidBrowser(){
  return /Android/i.test(navigator.userAgent||'');
}
function isIOSBrowser(){
  var ua=navigator.userAgent||'';
  // iPadOS 13+ reports a Macintosh UA, so a multi-touch "Mac" counts as iOS too
  return /iPhone|iPad|iPod/i.test(ua) || (navigator.maxTouchPoints>1 && /Macintosh/.test(ua));
}

function PhantomWalletAdapter(){
  this.name='Phantom';
  this.url='https://phantom.app';
  this.readyState=WalletReadyState.NotDetected;
  this.connecting=false;
  this.disconnecting=false;
  this._provider=null;
  this._publicKey=null;
  this._listeners={};
  this._eventsWired=false;
  this._connectPromise=null; // the ONE provider.connect() allowed in flight
  this._connectSilent=false;  // whether the in-flight connect is the silent auto-connect
}
PhantomWalletAdapter.prototype._getProvider=function(){
  var p=phantomProvider();
  if(p){ this._provider=p; this.readyState=WalletReadyState.Installed; }
  else { this.readyState=isMobileBrowser()?WalletReadyState.Loadable:WalletReadyState.NotDetected; }
  return this._provider;
};
Object.defineProperty(PhantomWalletAdapter.prototype,'publicKey',{ get:function(){ return this._publicKey; } });
Object.defineProperty(PhantomWalletAdapter.prototype,'connected',{ get:function(){ return !!(this._publicKey && this._provider && this._provider.isConnected); } });
PhantomWalletAdapter.prototype.on=function(evt, fn){ (this._listeners[evt]=this._listeners[evt]||[]).push(fn); return this; };
PhantomWalletAdapter.prototype.off=function(evt, fn){ var l=this._listeners[evt]; if(l){ var i=l.indexOf(fn); if(i>=0) l.splice(i,1); } return this; };
PhantomWalletAdapter.prototype.emit=function(evt, arg){ var l=(this._listeners[evt]||[]).slice(); for(var i=0;i<l.length;i++){ try{ l[i](arg); }catch(e){} } };
PhantomWalletAdapter.prototype._wireEvents=function(provider){
  if(this._eventsWired || !provider || !provider.on) return;
  this._eventsWired=true;
  var self=this;
  provider.on('accountChanged', function(key){
    self._publicKey=key||null;
    self.emit('accountChanged', self._publicKey);
  });
  provider.on('disconnect', function(){
    self._publicKey=null;
    self.emit('disconnect');
  });
};
// connect(opts) — opens Phantom's own connection prompt. Passing
// { onlyIfTrusted:true } is the adapter's silent auto-connect: it
// resolves without any prompt when the site is already trusted.
// SERIALIZED (the connect-freeze fix): Phantom allows only ONE
// approval window at a time. The silent auto-connect on page load used
// to overlap a manual tap, firing two provider.connect() calls at
// once — the approval prompt froze with no error and no way out. Now
// only one provider.connect() is EVER in flight: a second caller gets
// the same pending promise back. The one exception: a full (prompting)
// connect requested while only the silent auto-connect is in flight
// waits for the silent one to settle, then runs for real — still never
// two requests at once.
// TIMED: provider.connect() is raced against a 30s timeout, so a prompt
// that never answers rejects with code CONNECT_TIMEOUT and the UI can
// offer a clean "tap Connect Wallet to retry" instead of wedging on
// "Connecting…" forever.
var CONNECT_TIMEOUT_MS=30000;
PhantomWalletAdapter.prototype.connect=function(opts){
  var self=this;
  var silent=!!(opts&&opts.onlyIfTrusted);
  if(this._connectPromise){
    if(silent || !this._connectSilent) return this._connectPromise; // join the request already in flight
    // full connect requested while only the silent one runs: chain after it settles
    return this._connectPromise.then(function(){ return self.connect(opts); }, function(){ return self.connect(opts); });
  }
  var p=(async function(){
    var provider=self._getProvider();
    if(!provider){ var err=new Error('Phantom wallet not installed'); err.code='WALLET_NOT_INSTALLED'; throw err; }
    self._wireEvents(provider);
    if(self._publicKey) return;
    self.connecting=true;
    try{
      var res=await Promise.race([
        provider.connect(opts||undefined),
        new Promise(function(_, reject){
          setTimeout(function(){
            var te=new Error('Phantom didn\'t answer the connection request in time');
            te.code='CONNECT_TIMEOUT';
            reject(te);
          }, CONNECT_TIMEOUT_MS);
        })
      ]);
      self._publicKey=(res&&res.publicKey)||provider.publicKey||null;
      self.emit('connect', self._publicKey);
    }catch(e){ self.emit('error', e); throw e; }
    finally{ self.connecting=false; }
  })();
  this._connectPromise=p;
  this._connectSilent=silent;
  var clearPending=function(){ if(self._connectPromise===p){ self._connectPromise=null; self._connectSilent=false; } };
  p.then(clearPending, clearPending);
  return p;
};
PhantomWalletAdapter.prototype.disconnect=function(){
  var self=this;
  return (async function(){
    var provider=self._getProvider();
    self.disconnecting=true;
    try{ if(provider && provider.isConnected) await provider.disconnect(); }catch(e){}
    self._publicKey=null; self.disconnecting=false;
    self.emit('disconnect');
  })();
};
PhantomWalletAdapter.prototype.signTransaction=function(tx){
  var provider=this._getProvider();
  if(!provider||!this._publicKey) return Promise.reject(new Error('Wallet not connected'));
  return provider.signTransaction(tx);
};
PhantomWalletAdapter.prototype.signAllTransactions=function(txs){
  var provider=this._getProvider();
  if(!provider||!this._publicKey) return Promise.reject(new Error('Wallet not connected'));
  if(provider.signAllTransactions) return provider.signAllTransactions(txs);
  var self=this, out=[];
  return txs.reduce(function(p, tx){ return p.then(function(){ return self.signTransaction(tx).then(function(s){ out.push(s); }); }); }, Promise.resolve()).then(function(){ return out; });
};
PhantomWalletAdapter.prototype.signAndSendTransaction=function(tx){
  var provider=this._getProvider();
  if(!provider||!this._publicKey) return Promise.reject(new Error('Wallet not connected'));
  if(!provider.signAndSendTransaction) return Promise.reject(new Error('signAndSendTransaction unsupported by provider'));
  return provider.signAndSendTransaction(tx).then(function(res){ return (res&&res.signature)||res; });
};
PhantomWalletAdapter.prototype.signMessage=function(msgBytes){
  var provider=this._getProvider();
  if(!provider||!this._publicKey) return Promise.reject(new Error('Wallet not connected'));
  return provider.signMessage(msgBytes,'utf8');
};

function derivePdas(matchId){

// phantomSend — the ONE choke point for every Phantom-signed transaction
// (initialize_match, settle_match, cancel_match). Phantom-only: the
// provider is injected by Phantom's built-in browser, so every
// transaction signs natively in place via signAndSendTransaction —
// no app switching, no redirects, ever.
async function phantomSend(ixs){
  var conn=chainConn();
  var tx=new solanaWeb3.Transaction(); ixs.forEach(function(ix){ tx.add(ix); });
  tx.feePayer=chain.playerKey;
  var bh=await conn.getLatestBlockhash('confirmed');
  tx.recentBlockhash=bh.blockhash; tx.lastValidBlockHeight=bh.lastValidBlockHeight;
  if(!phantomProvider()) throw new Error('Phantom wallet not available — open SolSwing inside the Phantom app browser.');
  var sig;
  try{
    sig=await walletAdapter.signAndSendTransaction(tx); // adapter → Phantom signs + broadcasts
  }catch(e){
    if(!/unsupported|not supported|not implemented/i.test((e&&e.message)||'')) throw e; // a rejection in Phantom aborts — never retried silently
    var signed=await walletAdapter.signTransaction(tx);
    sig=await conn.sendRawTransaction(signed.serialize());
  }
  try{
    await conn.confirmTransaction({ signature:sig, blockhash:bh.blockhash, lastValidBlockHeight:bh.lastValidBlockHeight }, 'confirmed');
  }catch(ce){
    // A confirmation timeout is NOT a failed transaction — on a slow
    // connection (or a lagging RPC) the tx often lands anyway. Poll its
    // real status before reporting failure, so a landed escrow is never
    // announced as "no SOL moved".
    var landed=false;
    for(var pi=0;pi<8&&!landed;pi++){
      try{ var st2=await conn.getSignatureStatus(sig); if(st2&&st2.value&&!st2.value.err&&st2.value.confirmationStatus) landed=true; }catch(e2){}
      if(!landed) await new Promise(function(r){ setTimeout(r,1000); });
    }
    if(!landed) throw ce;
    log('[chain] tx confirmed via status poll after a slow confirmation · '+shortSig(sig),'winlog');
  }
  return sig;
}

}

// Fund the house opponent's stake with FREE devnet SOL (public faucet
// airdrop into its in-page wallet). Player funds are never used for this.
async function fundBot(needLamports){

  var bal=await conn.getBalance(bot.publicKey);
  if(bal>=needLamports) return true;
  var amt=(needLamports-bal)+20000000; // shortfall + 0.02 SOL fee buffer

  for(var attempt=0; attempt<3; attempt++){

      var sig=await conn.requestAirdrop(bot.publicKey, amt);
      await conn.confirmTransaction(sig,'confirmed');
      bal=await conn.getBalance(bot.publicKey);
