// The PIN is a device-local lock, NOT the money's security: only the
// Phantom wallet signature moves SOL. The PIN stops someone who grabs an
// unlocked phone from opening your account and playing. It is stored as a
// SHA-256 hash (PIN + wallet as salt), never as plain digits, never sent

async function pinHash(pin, wallet){
  var data=new TextEncoder().encode(pin+'|'+wallet);
  var digest=await crypto.subtle.digest('SHA-256', data);
  return Array.from(new Uint8Array(digest)).map(function(b){ return b.toString(16).padStart(2,'0'); }).join('');
}
function pinBytesToHex(bytes){ return Array.from(bytes).map(function(b){ return b.toString(16).padStart(2,'0'); }).join(''); }

    .catch(function(){

      if(realMode() && chain.escrowSig && !chain.joined && !chain.cancelSig){

      } else {
        setStatus('✋ Stake not completed — no SOL moved, no match started.');
      }
      log('[wallet] stake flow aborted','ruglog');
    });
}

async function pvpStakeTx(wagerSol, serverMatchId, adjudicatorStr){
  var lamports=Math.round(wagerSol*1e9);
  if(chain.escrowSig && !chain.joined && !chain.cancelSig && !chain.settleSig && phantomProvider()){
    try{ await chainCancel(); }catch(e){ /* previous escrow stays locked; the new stake still proceeds */ }
  }
  chain.matchId=BigInt(serverMatchId);
  var pdas=derivePdas(chain.matchId); chain.matchState=pdas.matchState; chain.vault=pdas.vault;
  chain.joined=false; chain.escrowSig=null; chain.joinSig=null; chain.settleSig=null; chain.cancelSig=null; chain.pendingSettle=null;
  var bal=await chainConn().getBalance(chain.playerKey);
  chain.balanceLamports=bal; updateFaucetNote();
  if(bal < lamports+3000000) throw new Error('insufficient');
  var pid=new solanaWeb3.PublicKey(PROGRAM_ID_STR);
  var adj=new solanaWeb3.PublicKey(adjudicatorStr);
  var ix=new solanaWeb3.TransactionInstruction({
    programId:pid,
    keys:[
      { pubkey:chain.matchState, isSigner:false, isWritable:true },
      { pubkey:chain.vault, isSigner:false, isWritable:true },
      { pubkey:chain.playerKey, isSigner:true, isWritable:true },
      { pubkey:adj, isSigner:false, isWritable:false }, // adjudicator: the server key — only the server can settle this match
      { pubkey:solanaWeb3.SystemProgram.programId, isSigner:false, isWritable:false }
    ],
    data:ixData(DISC.initialize_match, [u64le(chain.matchId), u64le(lamports)])
  });
  var sig=await phantomSend([ix]);
  chain.escrowSig=sig; chain.lastTx=sig;
  log('[chain] online initialize_match confirmed · match '+chain.matchId.toString()+' · escrowed '+fmt(wagerSol)+' SOL · tx '+shortSig(sig),'winlog');
  refreshBalance().then(updateFaucetNote);
  return sig;
}

    if(e && e.message==='insufficient') setStatus('⚠ Not enough devnet SOL for this wager + fees.');
    else setStatus('✋ Stake not completed — no SOL moved, no match started.');

  var pmT=pm<0?'no sell':pm+'ms', omT=om<0?'no sell':om+'ms';

function onlineEnd(m){
  var won=m.outcome==='win';
  var w=mmWager!==null?mmWager:wager();
  recordWager(w); // rewards: every staked SOL counts toward points + levels
  var payout=won?parseFloat(m.payout||0):0;
  recordMatchResult(won, payout, onlineBestMs); // leaderboard: W/L, P&L, fastest reflex
  var score=wins+' – '+losses;
  var oppLbl=onlineOpp;
  var settleTx=m.settleTx||null;
  onlineCleanup();
  $('btn-start').classList.remove('hidden');
  openMatchResult({ won:won, pnl:payout-w, wagered:w, payout:payout, score:score, note:'', voided:false, mode:'pvp', oppLabel:oppLbl, settleTx:settleTx });
  wins=0; losses=0; dots();
  var cs=$('chart-score'); if(cs) cs.textContent='0 – 0';
  var lbl=$('chart-score-label'); if(lbl) lbl.textContent='Match score · You – Bot';
  log('[online] match end · '+(won?'WIN':'LOSS')+' · payout '+fmt(payout)+' SOL','winlog');
}
function onlineCancelled(m){
  var wasCreatorStaked=realMode() && chain.escrowSig && !chain.joined && !chain.cancelSig;
  onlineCleanup();
  $('btn-start').classList.remove('hidden');
  if(wasCreatorStaked){
    setStatus('⏳ Match cancelled — refunding your stake on-chain…');
    chainCancel().then(function(){ setStatus('✓ Match cancelled — wager refunded on-chain. Queue again when ready.'); })
      .catch(function(){ setStatus('⚠ Match cancelled — refund failed; your escrow stays in the vault; leaving and re-entering the game screen retries the refund.'); });
  } else {
    setStatus('✋ Match cancelled — no SOL moved. Queue again when ready.');
  }
  log('[online] match cancelled · '+(m.reason||''),'ruglog');
}

}
function showMmCountdown(){
  var ov=$('mm-countdown'); if(!ov) return;
  mmActive=true;
  $('mm-you-bet').textContent=fmt(mmWager)+' SOL · LOCKED';
  $('mm-opp-name').textContent='🔴 '+oppName;
  $('mm-opp-bet').textContent=fmt(mmWager)+' SOL · LOCKED';
  $('mm-pot').textContent=fmt(mmWager*2)+' SOL';
  $('mm-note').textContent='Both bets are locked in the pot. Round 1 starts at zero.';
  $('mm-disconnect').classList.remove('hidden');
  ov.classList.add('open');
  mmDeadline=Date.now()+5000;
  // Simulated line drop: occasionally the opponent's connection dies mid-
  // countdown (production: the server's opponent_left message) — handled by
  // opponentDropped(), which cancels and returns YOU to the lobby.
  mmOppDropAt=(Math.random()<0.15) ? Date.now()+1600+Math.random()*2200 : 0;
  var numEl=$('mm-num'); numEl.style.color='';
  function tick(){
    if(!mmActive) return;
    if(!$('screen-game').classList.contains('active')||!walletConnected){ cleanupMatchmaking(); $('btn-start').classList.remove('hidden'); log('[matchmaking] countdown cancelled — player left · wager refunded','ruglog'); return; } // you disconnected mid-countdown
    if(mmOppDropAt && Date.now()>=mmOppDropAt){ opponentDropped(); return; }
    var remain=Math.ceil((mmDeadline-Date.now())/1000);
    if(remain<=0){ // zero → round 1 begins automatically
      mmActive=false; if(mmTimer){ clearInterval(mmTimer); mmTimer=null; }
      ov.classList.remove('open'); // wager stays locked — startRound owns it now, disconnectSession frees it at match end
      if(realMode() && !chain.joined){ // lock the opponent's stake on-chain (join_match) before round 1
        setStatus('🔒 Locking the opponent stake on-chain…');
        chainBotJoin(mmWager).then(function(){
          log('[matchmaking] countdown complete · round 1 starting · pot '+fmt(mmWager*2)+' SOL locked on-chain');
          startRound();
        }).catch(function(){
          setStatus('⚠ Opponent stake failed on-chain — refunding your wager…');
          chainCancel().catch(function(){}).then(function(){ lockWager(false); showScreen('lobby'); });
        });
        return;
      }
      log('[matchmaking] countdown complete · round 1 starting · pot '+fmt(mmWager*2)+' SOL');
      startRound(); return;
    }
    if(numEl.textContent!==String(remain)){ numEl.textContent=remain; numEl.classList.remove('pop'); void numEl.offsetWidth; numEl.classList.add('pop'); }
  }
  tick();
  mmTimer=setInterval(tick,200);
}
