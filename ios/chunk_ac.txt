 due —
  // it wins over the live chain fields, so a settle retried after a NEW
  // match has started still settles the match it actually belongs to.
  var ms=(ctx&&ctx.matchState)?new solanaWeb3.PublicKey(ctx.matchState):chain.matchState;
  var v=(ctx&&ctx.vault)?new solanaWeb3.PublicKey(ctx.vault):chain.vault;
  var winnerKey=(ctx&&ctx.winnerKey)?new solanaWeb3.PublicKey(ctx.winnerKey):((winnerId===1) ? chain.playerKey : chain.botKey.publicKey);
  var feeKey=new solanaWeb3.PublicKey(TREASURY_PUBKEY); // the program's locked fee receiver — account key only, never displayed
  return new solanaWeb3.TransactionInstruction({
    programId:pid,
    keys:[
      { pubkey:ms, isSigner:false, isWritable:true },
      { pubkey:v, isSigner:false, isWritable:true },
      { pubkey:winnerKey, isSigner:false, isWritable:true },
      { pubkey:feeKey, isSigner:false, isWritable:true },
      { pubkey:chain.playerKey, isSigner:true, isWritable:false } // adjudicator signs
    ],
    data:ixData(DISC.settle_match, [new Uint8Array([winnerId])])
  });
}

function disconnectSession(){
  clearTimers(); hideRoundCountdown(false); cancelPregameCountdown(); cleanupMatchmaking();
  lockWager(false); mmWager=null; // match over — the selector is free again, and no stale PvP stake can leak into the next match's math
  if(matchEndTimer){ clearTimeout(matchEndTimer); matchEndTimer=null; }
  active=false; rugged=false; youMs=null; botMs=null; rugStamp=0;
  wins=0; losses=0; roundNo=1; dots();
  oppName='Swing Bot'; oppShort='Bot'; oppSnap=false; telMatch={you:[],opp:[]}; roundTel=null;
  var dlbl=$('chart-score-label'); if(dlbl) dlbl.textContent='Match score · You – Bot';

}
function telCommit(who){
  var yRec={rt:(youMs!=null?youMs:null), sold:(youMs!=null), inputs:[], moves:0, pre:0};
  if(roundTel){
    yRec.moves=roundTel.moves;
    for(var i=0;i<roundTel.events.length;i++){ var d=roundTel.events[i]-rugStamp; if(d<0) yRec.pre++; else yRec.inputs.push(d); }
    roundTel=null;
  }
  if(yRec.sold&&yRec.inputs.length===0) yRec.inputs.push(yRec.rt);
  telMatch.you.push(yRec);
  var oSold=(who==='bot');
  var oRec={rt:(oSold?botMs:null), sold:oSold, inputs:[], moves:0, pre:0};
  if(oSold){ oRec.inputs.push(botMs); oRec.moves=oppSnap?0:Math.round(12+Math.random()*48); }
  telMatch.opp.push(oRec);
}
function reviewWinner(side){
  var all=telMatch[side]||[], recs=[], i;
  for(i=0;i<all.length;i++){ if(all[i].sold&&all[i].rt!=null) recs.push(all[i]); }
  var out={voided:false, score:0};
  if(recs.length<2) return out;
  var rts=recs.map(function(r){ return r.rt; });
  var mean=0, mv=0, k;
  for(k=0;k<rts.length;k++) mean+=rts[k];
  mean/=rts.length;
  var varSum=0; for(k=0;k<rts.length;k++) varSum+=(rts[k]-mean)*(rts[k]-mean);
  var sd=Math.sqrt(varSum/rts.length), cv=mean>0?sd/mean:1;
  var mx=Math.max.apply(null,rts), mn=Math.min.apply(null,rts);
  for(k=0;k<recs.length;k++) mv+=recs[k].moves;
  mv/=recs.length;
  var fast=(mean<100&&mx<130), steady=(cv<0.08&&(mx-mn)<25), still=(mv<2);
  out.score=(fast?45:0)+(steady?30:0)+(still?25:0);
  out.voided=(fast&&steady&&still);
  return out;
}
function endMatch(won, note){
  var w=(matchWager!==null?matchWager:wager()); // this match's stake per player
  var pot=w*2;
  var finalScore=wins+' – '+losses; // captured BEFORE the session state is wiped
  var oppLbl=oppShort; // likewise captured before the wipe
  // Settlement shape mirrors the adjudicator's settle call: one share per
  // side — winner side, loser side, house side — always summing to the pot.
  var verdict=reviewWinner(won?'you':'opp');
  var settle={winner:(won?'you':'opp'), voided:verdict.voided, score:verdict.score,
    winnerShare:pot*0.94, loserShare:0, houseShare:pot*0.06};
  if(verdict.voided){ settle.winnerShare=0; settle.loserShare=pot*0.5; settle.houseShare=pot*0.5; }
  var payout=won? settle.winnerShare : settle.loserShare; // what lands back in YOUR wallet
  var pnl=payout-w;
  // REAL on-chain matches: the anti-bot review stays STAGED (it never moves
  // money) — settle_match only knows winner 1 or 2, so a flagged result
  // settles the pot to the NON-flagged side, 94/6 like any other match.
  var isRealChain=realMode() && chain.joined && !!chain.escrowSig;
  if(isRealChain && verdict.voided){
    var flaggedYou=won; // the review inspected the winner — that's who got flagged
    settle.winner=flaggedYou?'opp':'you'; settle.winnerShare=pot*0.94; settle.loserShare=0; settle.houseShare=pot*0.06;
    payout=flaggedYou?0:pot*0.94; pnl=payout-w; verdict.voided=false; verdict.chainVoid=true;
  }
  recordMatch(won&&!verdict.voided&&!verdict.chainVoid, payout); // a flagged "win" the review overturned never counts as a won match in the stats
  if(verdict.voided){
    log('[server] match_end outcome=void · settle_match → adjudicator · winner share 0.000 SOL · loser refunded '+fmt(settle.loserShare)+' SOL', won?'ruglog':'winlog');
    log(treasuryLockOk()?'[chain] house share '+fmt(settle.houseShare)+' SOL → treasury wallet · LOCKED ✓ (the one locked fee receiver)':'[chain] ⛔ TREASURY LOCK CHECK FAILED — settlement refused','winlog');
  }
  else if(won){ level++; log('[server] match_end outcome=win payout='+fmt(payout)+' SOL · settle_match → adjudicator','winlog'); log(treasuryLockOk()?'[chain] fee '+fmt(settle.houseShare)+' SOL (6%) → treasury wallet · LOCKED ✓ (the one locked fee receiver)':'[chain] ⛔ TREASURY LOCK CHECK FAILED — settlement refused','winlog'); }
  else { if(level>1) level--; log('[server] match_end outcome=loss','ruglog'); }
  $('rank').textContent='Level '+level+' · '+((won&&!verdict.voided)?'CHAD':'JEET');
  if(walletDemo){ demoBalance=Math.max(0, demoBalance+pnl); balanceText=fmt(demoBalance)+' SOL · demo'; }
  disconnectSession();   // 1 · disconnect immediately — session gone, state wiped
  showScreen('lobby');   // 2 · relocate to the previous screen (the lobby)
  if(!won){
    // 3 · the waiting player watches the puzzle popup while the winner is
    // checked; it closes itself on a pass (normal result follows) and turns
    // into the refund announcement on a confirmed catch.
    openVerifyPopup();
    var lossNote=note||'';
    setTimeout(function(){
      if(verdict.voided){ openRefundResult({refund:settle.loserShare, wagered:w, score:finalScore, opp:oppLbl}); }
      else { closeRefundResult(true); openMatchResult({won:false, pnl:pnl, wagered:w, payout:payout, score:finalScore, note:lossNote, voided:false}); }
    }, 2600+Math.random()*900);
  }
  else { openMatchResult({won:won, pnl:pnl, wagered:w, payout:payout, score:finalScore, note:note||'', voided:verdict.voided}); } // 3 · stats popup on arrival
  // 4 · REAL settlement: the adjudicator (your wallet) signs settle_match —
  // the program pays 94% to the winner / 6% to the locked treasury wallet.
  // The owed settle is recorded from the CHAIN state (joined + escrowed +
  // unsettled), not from the wallet's current connection: if the wallet
  // dropped mid-match, the pot is still locked on-chain and the settle is
  // still owed — it survives the disconnect and is offered again the
  // moment the wallet reconnects (see recoverOwedSettle). The account set
  // is snapshotted NOW, so a settle retried later can never land on a
  // different match. (isRealChain above still gates the anti-bot verdict
  // adjustments, which need a live connection to mean anything.)
  var settleOwed=chain.joined && !!chain.escrowSig && !chain.settleSig;
  if(settleOwed){
    // summary rides along so a settle retried later rebuilds this exact
    // result popup.
    var winKey=(settle.winner==='you') ? chain.playerKey : (chain.botKey&&chain.botKey.publicKey);
    chain.pendingSettle={ winnerId:(settle.winner==='you'?1:2),
      accounts:{ matchState:chain.matchState?chain.matchState.toString():null, vault:chain.vault?chain.vault.toString():null, winnerKey:winKey?winKey.toString():null },
      summary:{ won:won, pnl:pnl, wagered:w, payout:payout, score:finalScore, note:note||'', voided:false, mode:matchMode } };
    if(realMode()) chainDoSettle();
    else log('[chain] settle_match still owed — reconnect the wallet to sign it and release the pot','ruglog');
  }
}

// WALLET SIGNATURE GUARD — HARD SECURITY RULE.
// No SOL can EVER move without the player's own wallet signature. Every
// stake (and, on the real on-chain build, every payout confirmation)
// routes through requireWalletSignature(), which resolves ONLY after the
// player signs — Phantom signMessage for a connected wallet, an explicit
// sign tap in this window for the demo wallet. Cancel, reject, X,
// tap-outside or Escape reject the promise and the caller aborts: the
// match never starts and zero funds move. Nothing auto-signs, ever.
// ═══════════════════════════════════════════════════════════════════════
var signPending=null;
function requireWalletSignature(actionLabel, detail, amountSol){
  return new Promise(function(resolve, reject){
    if(signPending){ var old=signPending; signPending=null; old.reject(new Error('superseded')); }
    signPending={resolve:resolve, reject:reject, kind:'message'};
    $('sg-action').textContent=String(actionLabel).toUpperCase();
    $('sg-detail').textContent=detail;
    $('sg-amount').textContent=fmt(amountSol)+' SOL';
    $('sg-signed').textContent='✓ Signed by your wallet';
    $('sg-signed').classList.add('hidden');
    var trx=$('sg-txrow'); if(trx) trx.classList.add('hidden');
    $('sg-sign').textContent='✍️ Sign with Wallet';
    $('sg-sign').disabled=false;
    $('sign-modal').classList.add('open');
    $('sg-sign').focus();
  });
}
// The stake gate: with Phantom connected this sends the REAL
// initialize_match transaction (your SOL escrows into the program vault
// only when Phantom confirms it); with the demo wallet it falls back to
// the signature-message guard above. Resolves only after the escrow is
// confirmed on-chain (real mode) or the signature lands (demo mode).
function requireStake(detail, wagerSol){
  if(!realMode()) return requireWalletSignature('Stake', detail, wagerSol);
  return new Promise(function(resolve, reject){
    if(signPending){ var old=signPending; signPending=null; old.reject(new Error('superseded')); }
    signPending={resolve:resolve, reject:reject, kind:'stake', wagerSol:wagerSol};
    $('sg-action').textContent='STAKE · ON-CHAIN';
    $('sg-detail').textContent=detail+' Phantom will ask you to approve the initialize_match transaction that escrows it into the SolSwing program vault on devnet.';
    $('sg-amount').textContent=fmt(wagerSol)+' SOL';
    $('sg-signed').classList.add('hidden');
    var trx=$('sg-txrow'); if(trx) trx.classList.add('hidden');
    $('sg-sign').textContent='✍️ Sign & Stake in Phantom';
    $('sg-sign').disabled=false;
    $('sign-modal').classList.add('open');
    $('sg-sign').focus();
  });
}
function settleSignature(okSig){
  var p=signPending; signPending=null;
  $('sign-modal').classList.remove('open');
  if(p){ if(okSig) p.resolve(true); else p.reject(new Error('signature declined')); }
}
$('sg-sign').addEventListener('click', async function(){
  if(signPending && signPending.kind==='stake'){ // REAL on-chain stake: initialize_match via Phantom
    var sbtn=this; sbtn.disabled=true;
    var myPending=signPending, myWager=signPending.wagerSol; // identity token: everything below must only ever resolve THIS request
    try{
      var esig=await chainStake(myWager);
      if(signPending!==myPending){
        // Cancelled / superseded / wallet dropped WHILE Phantom was
        // signing — but the escrow transaction landed. Refund it now:
        // never leave SOL locked in the vault behind a "stake not
        // completed" message.
        if(chain.escrowSig===esig && !chain.joined && !chain.cancelSig){
          setStatus('⚠ Stake landed after the request was cancelled — refunding your escrow on-chain…');
          chainCancel().then(function(){ setStatus('✓ Escrow refunded on-chain — no match started, no SOL kept.'); }).catch(function(){ setStatus('⚠ Escrow refund could not be sent — your SOL stays in the match vault; reconnect and leave the game screen to retry the refund.'); });
          log('[chain] post-cancel escrow refund fired · tx '+shortSig(esig),'ruglog');
        }
        return;
      }
      var trow=$('sg-txrow'), tEl=$('sg-tx');
      if(trow&&tEl){ tEl.innerHTML='<a href="'+explorerTx(esig)+'" target="_blank" rel="noopener">'+shortSig(esig)+' ↗</a>'; trow.classList.remove('hidden'); }
      var sEl=$('sg-signed'); sEl.textContent='✓ Escrow confirmed on-chain'; sEl.classList.remove('hidden');
      setTimeout(function(){ if(signPending===myPending) settleSignature(true); }, 900);
    }catch(e){
      if(signPending!==myPending){ return; } // already resolved by the cancel/supersede — never touch the newer request's state
      if(e && e.message==='insufficient'){
        // Keep the window open with the faucet directions — the player can
        // top up free devnet SOL and hit Sign again, or cancel out.
        $('sg-detail').textContent='⚠ Not enough devnet SOL for this wager + rent/fees. Devnet SOL is free: get some from faucet.solana.com (it asks for a GitHub sign-in), then hit Sign & Stake again — or Cancel to back out. Nothing has moved.';
        sbtn.disabled=false;
      } else { settleSignature(false); } // rejected in Phantom / tx failed → abort, no match
    }
    return;
  }
  this.disabled=true;
  var myPendingMsg=signPending;
  try{
    if(walletConnected && !walletDemo && walletAdapter.connected){
      var msg='SolSwing · '+$('sg-action').textContent+' · '+$('sg-amount').textContent+' · '+new Date().toISOString();
      await walletAdapter.signMessage(new TextEncoder().encode(msg)); // real Phantom signature, via the adapter
    }
    if(signPending!==myPendingMsg) return; // cancelled while Phantom was open — the abort already resolved this request; a late signature must not resolve a newer one
    $('sg-signed').classList.remove('hidden'); // ✓ Signed by your wallet
    setTimeout(function(){ if(signPending===myPendingMsg) settleSignature(true); }, 700);
  }catch(e){ if(signPending===myPendingMsg) settleSignature(false); } // player rejected in Phantom → abort, no funds move
});

function disconnectWallet(){
  cleanupMatchmaking(); // a wallet disconnect mid-queue/countdown cancels matchmaking
  if(signPending) settleSignature(false); // an armed signature request dies with the session — it must never resolve against a disconnected wallet
  // An open, unfilled on-chain escrow gets its cancel_match refund fired
  // BEFORE the wallet drops — the refund needs your signature.
  if(realMode() && phantomProvider() && chain.escrowSig && !chain.joined && !chain.cancelSig && !chain.settleSig){ var cp=chainCancel(); if(cp&&cp.catch) cp.catch(function(){}); }
  var wasPhantom=walletConnected && !walletDemo;
  // An unsettled JOINED match keeps its whole chain context across the
  // disconnect: its pot is locked on-chain and its settle is still owed.
  // Wiping it here would strand the pot with no in-page way back. Any
  // other chain state belongs to a finished or abandoned match — clear
  // it, so a stale escrow can never leak into the next wallet session.
  var settleOwedDc=!!(chain.pendingSettle && chain.joined && chain.escrowSig && !chain.settleSig);
  walletConnected=false; walletDemo=false; walletAddr=''; walletFullAddr=''; balanceText='— SOL'; demoBalance=10;
  chain.playerKey=null; chain.balanceLamports=null; if(!settleOwedDc) chain.pendingSettle=null; updateFaucetNote();
  if(!settleOwedDc){ chain.escrowSig=null; chain.joinSig=null; chain.settleSig=null; chain.cancelSig=null; chain.joined=false; chain.matchId=null; chain.matchState=null; chain.vault=null; }
  if(!active && matchWager===null) lockWager(false);
  // Disconnect through the adapter. Local state is already torn down, so
  // the adapter's 'disconnect' event listener no-ops instead of looping.
  if(wasPhantom){ try{ var dp=walletAdapter.disconnect(); if(dp&&dp.catch) dp.catch(function(){}); }catch(e){} }
  paintWalletButton();
  closeMatchResult();
  closeRefundResult();
  log('[wallet] disconnected');
}

function finishRound(who){
  if(!active) return; active=false; clearTimers();
  $('rug').classList.remove('active'); // round decided — clear the rug flash, result lives in the status line
  if(who==='you'&&youMs==null) youMs=Date.now()-rugStamp;
  if(botMs==null) botMs=9999;
  telCommit(who);
  $('btn-sell').classList.add('hidden'); $('btn-start').classList.remove('hidden'); $('btn-start').textContent='Queue Next Round';
  if(who==='you'){ wins++; setStatus('🟢 Frontrun! You sold in '+youMs+'ms vs '+oppShort+' '+botMs+'ms.'); log('[server] round_result winner=player · '+youMs+'ms vs '+botMs+'ms','winlog'); }
  else { losses++; if(youMs!=null){ setStatus('🔴 '+oppShort+' beat you — '+botMs+'ms vs your '+youMs+'ms.'); } else { setStatus('🔴 RUGGED! '+oppShort+' sold in '+botMs+'ms — you never exited.'); } log('[server] round_result winner=opponent · '+botMs+'ms','ruglog'); }
  dots(); roundNo++;
  if(wins>=2||losses>=2){ endMatch(wins>=2); } // decided → disconnect NOW: lobby + stats popup (see endMatch)
  else { showRoundCountdown(); } // match not decided — 5s countdown overlay, then the next round starts itself
  updateHUD();
}
function forfeit(reason){
  if(!active) return; active=false; clearTimers(); losses=2; dots();
  log('[server] forfeit · '+reason,'ruglog');
  endMatch(false, 'Forfeited'); // decided → disconnect NOW: lobby + stats popup
}

$('btn-start').addEventListener('click', function(){
  if(active||signPending||pregameActive||waiting||mmActive) return;
  if(!walletConnected){ openConnectModal(); return; } // no wallet → the connect popup, right here on the game screen
  if(walletDemo && wins===0 && losses===0 && demoBalance < wager()-1e-9){ setStatus('Demo wallet is short for a '+fmt(wager())+' SOL wager — disconnect the demo wallet and reconnect it to top back up to 10 SOL.'); return; } // a stake the balance can't cover is never signed (the real wallet path enforces the same rule on-chain via the 'insufficient' check)
  if(wins>0||losses>0){ startRound(); return; } // safety only — mid-match rounds auto-start via the countdown
  if(matchMode==='pvp'){ startMatchmaking(); return; } // PvP: sign → wait for opponent → both bets locked → matchmaking countdown → round 1
  startPregameCountdown();
});
