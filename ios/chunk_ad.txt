// WALLET SIGNATURE GUARD — HARD SECURITY RULE.
// No SOL can EVER move without the player's own wallet signature. Every
// stake (and, on the real on-chain build, every payout confirmation)
// routes through requireWalletSignature(), which resolves ONLY after the
// player signs — Phantom signMessage for a connected wallet, an explicit
// sign tap in this window for the demo wallet. Cancel, reject, X,
// tap-outside or Escape reject the promise and the caller aborts: the
// match never starts and zero funds move. Nothing auto-signs, ever.
// ═══════════════════════════════════════════════════════════════════════
var signPending=null;
function requireWalletSignature(actionLabel, detail, amountSol){
  return new Promise(function(resolve, reject){
    if(signPending){ var old=signPending; signPending=null; old.reject(new Error('superseded')); }
    signPending={resolve:resolve, reject:reject, kind:'message'};
    $('sg-action').textContent=String(actionLabel).toUpperCase();
    $('sg-detail').textContent=detail;
    $('sg-amount').textContent=fmt(amountSol)+' SOL';
    $('sg-signed').textContent='✓ Signed by your wallet';
    $('sg-signed').classList.add('hidden');
    var trx=$('sg-txrow'); if(trx) trx.classList.add('hidden');
    $('sg-sign').textContent='✍️ Sign with Wallet';
    $('sg-sign').disabled=false;
    $('sign-modal').classList.add('open');
    $('sg-sign').focus();
  });
}
// The stake gate: with Phantom connected this sends the REAL
// initialize_match transaction (your SOL escrows into the program vault
// only when Phantom confirms it); with the demo wallet it falls back to
// the signature-message guard above. Resolves only after the escrow is
// confirmed on-chain (real mode) or the signature lands (demo mode).
function requireStake(detail, wagerSol){
  if(!realMode()) return requireWalletSignature('Stake', detail, wagerSol);
  // No in-game transaction screen: Join Game goes straight to Phantom's own
  // approval popup. The stake still resolves ONLY after Phantom confirms the
  // escrow on-chain, and a Phantom decline still rejects so the caller aborts
  // with zero moved. Phantom's popup itself stays — that is where the player
  // approves spending, and it can never be skipped.
  return new Promise(function(resolve, reject){
    if(signPending){ var old=signPending; signPending=null; old.reject(new Error('superseded')); }
    var myPending={resolve:resolve, reject:reject, kind:'stake', wagerSol:wagerSol};
    signPending=myPending;
    Promise.resolve().then(function(){ return chainStake(wagerSol); }).then(function(esig){
      if(signPending!==myPending){
        // Cancelled / superseded / wallet dropped WHILE Phantom was signing —
        // but the escrow landed. Refund it now, never leave SOL locked behind
        // a "stake not completed" message.
        if(chain.escrowSig===esig && !chain.joined && !chain.cancelSig){
          setStatus('⚠ Stake landed after the request was cancelled — refunding your escrow on-chain…');
          chainCancel().then(function(){ setStatus('✓ Escrow refunded on-chain — no match started, no SOL kept.'); }).catch(function(){ setStatus('⚠ Escrow refund could not be sent — your SOL stays in the match vault; reconnect and leave the game screen to retry the refund.'); });
          log('[chain] post-cancel escrow refund fired · tx '+shortSig(esig),'ruglog');
        }
        return;
      }
      signPending=null;
      resolve(true);
    }).catch(function(e){
      if(signPending!==myPending){ return; } // already resolved by cancel/supersede — never touch the newer request
      signPending=null;
      if(e && e.message==='insufficient'){
        // No modal anymore to hold the faucet directions, so leave them as the
        // final status line after the caller's abort message lands.
        setTimeout(function(){ setStatus('⚠ Not enough devnet SOL for this wager + rent/fees. Get free devnet SOL from faucet.solana.com, then Join Game again. Nothing has moved.'); }, 0);
      }
      reject(e);
    });
  });
}

}

    // load); if it never appears, this page is open in a regular
    // browser — the gate is the answer, not a handoff.
    if(!phantomInstalled()){
      var found=await waitForPhantom(2600);
      if(!found){
        showPhantomGate();
        setCwState('err','Open SolSwing in Phantom','This game runs only inside the Phantom app\'s built-in browser. Open Phantom, tap the browser icon, and enter solswing.vercel.app.');
        var note=$('cw-install-note'); if(note) note.classList.remove('hidden');
        setStatus('Open SolSwing inside the Phantom app browser to connect your wallet.');
        return;
      }
      hidePhantomGate();
    }
    await walletAdapter.connect(); // window.phantom.solana.connect() — Phantom's native in-app prompt
    await syncConnectedState(); // address + REAL devnet balance + chain key
    onWalletConnected(); // green Connected panel + toast + green wallet button
    if(chain.balanceLamports!==null && chain.balanceLamports < Math.round(wager()*1e9)+3000000){
      setStatus('Wallet connected — but it can\'t cover this wager + fees yet. Devnet SOL is free: grab some from faucet.solana.com (GitHub sign-in), then queue.');
    } else {
      setStatus('Wallet connected · devnet balance '+balanceText+'. Queue a match to swing!');
    }
    updateHUD(); showScreen('game');
    setTimeout(closeConnectModal, 1600); // hold the green Connected state a beat, then step aside
  }catch(e){
    // Phantom rejection (code 4001) is a choice, not a failure — say so,
    // in the popup, one tap from trying again.
    openConnectModal();
    if(e&&e.code===4001){
      setCwState('err','Connection declined','You declined in Phantom — nothing was connected and no SOL moved. Tap Connect Wallet to try again.');
      setStatus('Connection declined in Phantom — tap Connect Wallet and approve to play for real.');
    } else if(e&&e.code==='CONNECT_TIMEOUT'){
      // The 30s race in the adapter fired: Phantom never answered. The
      // pending request is cleared, so this same button retries clean.
      setCwState('err','Connection timed out','Phantom didn\'t answer within 30 seconds — tap Connect Wallet to retry.');
      setStatus('Connection timed out — tap Connect Wallet to retry.');
    } else {
      setCwState('err','Connection didn\'t go through',((e&&e.message)?e.message+' — ':'')+'tap Connect Wallet to retry.');
      setStatus('Wallet connection didn\'t go through — tap Connect Wallet to try again.');
    }
  } finally {
    connectBusy=false;
    if(btn){ btn.disabled=false; btn.textContent='Connect Wallet'; } // the button is always the retry
  }
}

var chatSocket=null;
var CHAT_WS_URL='wss://solswing-server-production.up.railway.app/';
function chatConnect(){
  chatUserList();
  if(chatSocket && (chatSocket.readyState===WebSocket.OPEN || chatSocket.readyState===WebSocket.CONNECTING)) return;
  try{ chatSocket=new WebSocket(CHAT_WS_URL); }catch(e){ addChatSys('Live chat unavailable — messages stay on this device.'); return; }
  chatSocket.onopen=function(){ addChatSys('Connected to live arena chat.'); chatUserList(); };
  chatSocket.onmessage=function(ev){
    try{
      var m=JSON.parse(ev.data);
      if(m.type==='chat' && m.user && m.text){
        chatSeenUsers[m.user]=true; chatUserList();
        addChat(m.user, m.text, m.user===getUsername());
      } else if(m.type==='presence'){
        var p=$('chat-online'); if(p) p.textContent=m.count+' online';
      }
    }catch(e){}
  };
  chatSocket.onclose=function(){ chatSocket=null; };
}
$('chat-form').addEventListener('submit', function(e){
  e.preventDefault();
  var inp=$('chat-input'); var v=inp.value.trim(); if(!v) return;
  inp.value='';
  var me=getUsername();
  if(!me){ addChatSys('Pick an arena name on the first page before chatting.'); return; }
  if(chatSocket && chatSocket.readyState===WebSocket.OPEN){
    // No local echo: the server broadcasts back to the sender too, so your
    // message appears once, in the same stream as everyone else's.
    chatSocket.send(JSON.stringify({type:'chat', user:me, text:v.slice(0,240)}));
  } else {
    addChat(me, v, true);
    addChatSys('Live chat not connected — your message shows only here.');
    chatConnect();
  }
});
